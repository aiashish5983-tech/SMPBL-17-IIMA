      startTimer();

    }


    function selectOption(qNum, selectedKey) {

      userAnswers[qNum] = selectedKey;


      ['a', 'b', 'c', 'd'].forEach(k => {

        const label = document.getElementById(`label_${qNum}_${k}`);

        if (label) label.classList.remove("selected");

      });

      const chosenLabel = document.getElementById(`label_${qNum}_${selectedKey}`);

      if (chosenLabel) chosenLabel.classList.add("selected");


      const bubble = document.getElementById(`bubble_${qNum}`);

      if (bubble) bubble.classList.add("answered");


      const answeredCount = Object.keys(userAnswers).length;

      document.getElementById("progressCount").innerText = `${answeredCount} / 50`;

    }


    function startTimer() {

      const badge = document.getElementById("timerBadge");

      timerInterval = setInterval(() => {

        totalSeconds--;

        if (totalSeconds <= 0) {

          clearInterval(timerInterval);

          finishExam(true);

          return;

        }


        const mins = Math.floor(totalSeconds / 60);

        const secs = totalSeconds % 60;

        badge.innerText = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;


        if (totalSeconds <= 300) {

          badge.classList.add("warning");

        }

      }, 1000);

    }


    function confirmSubmit() {

      const answered = Object.keys(userAnswers).length;

      const unattempted = 50 - answered;

      const msg = unattempted > 0 

        ? `You have answered ${answered} questions. You still have ${unattempted} unanswered question(s).\n\nDo you want to submit and generate your final score card?`

        : "Are you sure you want to finish the exam and view your report?";

      if (confirm(msg)) {

        finishExam(false);

      }

    }


    function finishExam(isTimeOut) {

      clearInterval(timerInterval);


      if (isTimeOut) {

        alert("Time has expired! Your exam will now be scored.");

      }


      // Read student meta

      const nameVal = document.getElementById("studentNameInput").value.trim() || "Candidate";

      const rollVal = document.getElementById("studentRollInput").value.trim() || "N/A";

      const dateStr = new Date().toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' });

      document.getElementById("studentIdentityDisplay").innerText = `Student: ${nameVal} | Roll/ID: ${rollVal} | Date: ${dateStr}`;


      document.getElementById("quizContainer").style.display = "none";

      document.getElementById("headerControls").style.display = "none";

      document.getElementById("resultsScreen").style.display = "block";

      window.scrollTo({ top: 0, behavior: 'smooth' });


      let score = 0;

      const reviewContainer = document.getElementById("reviewFeed");

      reviewContainer.innerHTML = "";


      quizData.forEach((item, index) => {

        const qNum = index + 1;

        const userChoice = userAnswers[qNum] || null;

        const isCorrect = userChoice === item.answer;


        if (isCorrect) score++;


        const reviewCard = document.createElement("div");

        reviewCard.className = `review-item ${isCorrect ? 'correct-item' : 'incorrect-item'}`;

        

        let statusBadge = isCorrect 

          ? `<span style="color: var(--success); font-weight: 700;">✓ Correct</span>`

          : `<span style="color: var(--danger); font-weight: 700;">✗ ${userChoice ? 'Incorrect' : 'Unattempted'}</span>`;


        reviewCard.innerHTML = `

          <div style="display: flex; justify-content: space-between; margin-bottom: 8px;">

            <span style="font-weight: 700; color: var(--text-muted);">Question ${qNum}</span>

            ${statusBadge}

          </div>

          <div style="font-weight: 600; font-size: 1rem; margin-bottom: 12px;">${item.q}</div>

          <div style="display: flex; flex-direction: column; gap: 4px;">

            ${Object.entries(item.options).map(([k, text]) => {

              let optClass = "option-review";

              if (k === item.answer) optClass += " is-correct";

              if (userChoice === k && !isCorrect) optClass += " is-wrong-selection";

              

              let marker = "";

              if (k === item.answer) marker = " (Correct Answer)";

              if (userChoice === k && !isCorrect) marker = " (Your Choice)";

              if (userChoice === k && isCorrect) marker = " (Your Choice ✓)";


              return `<div class="${optClass}"><strong>${k.toUpperCase()}.</strong> ${text}${marker}</div>`;

            }).join("")}

          </div>

          <div class="explanation-box">

            <strong>Accounting Explanation:</strong> ${item.exp}

          </div>

        `;

        reviewContainer.appendChild(reviewCard);

      });


      const elapsed = (30 * 60) - totalSeconds;

      const elapsedMins = Math.floor(elapsed / 60);

      const elapsedSecs = elapsed % 60;


      document.getElementById("scoreDisplay").innerText = `${score} / 50`;

      document.getElementById("accuracyStat").innerText = `${Math.round((score / 50) * 100)}%`;

      document.getElementById("answeredStat").innerText = `${Object.keys(userAnswers).length} / 50`;

      document.getElementById("timeTakenStat").innerText = `${elapsedMins}m ${elapsedSecs.toString().padStart(2, '0')}s`;

    }


    window.onload = initQuiz;
